import java.util.*;

class StockSpan {

    static int[] calculateSpan(int[] price) {
        int n = price.length;
        int[] span = new int[n];
        Stack<Integer> st = new Stack<>();

        // First element span is always 1
        st.push(0);
        span[0] = 1;

        for (int i = 1; i < n; i++) {

            // Pop elements smaller or equal to current price
            while (!st.isEmpty() && price[st.peek()] <= price[i]) {
                st.pop();
            }

            // If stack is empty → all previous are smaller
            if (st.isEmpty()) {
                span[i] = i + 1;
            } else {
                span[i] = i - st.peek();
            }

            // Push current index
            st.push(i);
        }

        return span;
    }

    public static void main(String[] args) {
        int[] price = {100, 80, 60, 70, 60, 75, 85};

        int[] result = calculateSpan(price);

        System.out.println(Arrays.toString(result));
    }
}
